💾 DATABASE SCHEMA (Prisma)
Run this Prisma schema exactly as written. Do not modify:

prisma
model User {
  id              String    @id @default(cuid())
  email           String    @unique
  passwordHash    String
  role            Role      @default(STUDENT)
  fullName        String
  cnic            String?   @unique
  phone           String?
  emailVerified   DateTime?
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt
  
  // Relations
  studentProfile  StudentProfile?
  counselorProfile CounselorProfile?
  adminProfile    AdminProfile?
  applications    Application[]
  recommenders    Recommender[]
}

model StudentProfile {
  id                String   @id @default(cuid())
  userId            String   @unique
  dateOfBirth       DateTime?
  gender            Gender?
  nationality       String   @default("Pakistani")
  domicileProvince  String?
  domicileDistrict  String?
  religion          String?
  languages         String[]   // Store as JSON or separate table
  disabilityStatus  String?
  currentAddress    String?
  permanentAddress  String?
  guardianName      String?
  guardianRelation  String?
  
  // Family
  fatherName        String?
  fatherOccupation  String?
  fatherEducation   String?
  motherName        String?
  motherOccupation  String?
  motherEducation   String?
  siblings          Int?
  familyIncome      Int?       // Annual income in PKR
  
  // Education
  matricSchool      String?
  matricBoard       String?
  matricYear        Int?
  matricMarks       String?    // e.g., "850/1100"
  matricGrade       String?
  interCollege      String?
  interBoard        String?
  interYear         Int?
  interMarks        String?
  interGrade        String?
  hecEquivalence    String?
  
  // Testing
  testType          String?    // "NAT", "University Specific"
  testRegNo         String?
  testDate          DateTime?
  testScores        Json?      // Flexible JSON for different test types
  englishProficiency String?
  
  // Status
  applicationStatus  ApplicationStatus @default(INCOMPLETE)
  
  userId_User User @relation(fields: [userId], references: [id])
}

model University {
  id           String   @id @default(cuid())
  name         String   @unique
  city         String
  province     String
  programs     String[]
  deadline     DateTime?
  applicationFee Int?
  testRequired Boolean  @default(true)
  website      String?
  createdAt    DateTime @default(now())
  
  supplementaryQuestions SupplementaryQuestion[]
  applications  Application[]
}

model SupplementaryQuestion {
  id             String   @id @default(cuid())
  universityId   String
  questionText   String
  questionType   QuestionType @default(TEXTAREA)
  maxLength      Int?      // Character limit
  isRequired     Boolean  @default(true)
  orderIndex     Int      @default(0)
  
  university University @relation(fields: [universityId], references: [id])
  answers     SupplementaryAnswer[]
}

model Application {
  id            String   @id @default(cuid())
  userId        String
  universityId  String
  status        SubmissionStatus @default(DRAFT)
  submittedAt   DateTime?
  feePaid       Boolean  @default(false)
  feePaymentId  String?
  totalScore    Float?      // For ranking purposes
  
  user    User        @relation(fields: [userId], references: [id])
  university University @relation(fields: [universityId], references: [id])
  answers  SupplementaryAnswer[]
}

model SupplementaryAnswer {
  id                   String   @id @default(cuid())
  applicationId        String
  supplementaryQuestionId String
  answerText           String   @db.Text
  
  application Application @relation(fields: [applicationId], references: [id])
  question SupplementaryQuestion @relation(fields: [supplementaryQuestionId], references: [id])
}

model Recommender {
  id            String   @id @default(cuid())
  userId        String
  name          String
  email         String
  designation   String    // "Counselor", "Teacher", "Other"
  relationship  String
  phone         String?
  invitationSent DateTime?
  invitationStatus InvitationStatus @default(PENDING)
  submittedAt   DateTime?
  recommendationText String? @db.Text
  
  user User @relation(fields: [userId], references: [id])
}

enum Role {
  STUDENT
  COUNSELOR
  ADMIN
}

enum Gender {
  MALE
  FEMALE
  OTHER
}

enum ApplicationStatus {
  INCOMPLETE
  COMPLETE
  SUBMITTED
}

enum SubmissionStatus {
  DRAFT
  SUBMITTED
  UNDER_REVIEW
  ACCEPTED
  REJECTED
}

enum QuestionType {
  TEXT
  TEXTAREA
  SELECT
  CHECKBOX
}

enum InvitationStatus {
  PENDING
  SENT
  COMPLETED
}
